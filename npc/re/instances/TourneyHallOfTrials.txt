//===== rAthena Script =======================================
//= Instance: Tourney Hall of Trials.
//===== Description: =========================================
//- [Walkthrough conversion]
//- Part of the Imperial Guard job change (npc/re/jobs/4-2/imperial_guard.txt).
//- Dialog, opponent appearances and NPC sprites follow the iRO
//- gameplay video https://www.youtube.com/watch?v=FUaZyJYUa6U.
//- Branches not shown there use the official client's English strings.
//- Two one-on-one rounds against the Knight of Devotion and the
//- Knight of Armament; the order comes from the quest the King's
//- Knight gave in the Palace (3438: Devotion first, 3436: Armament
//- first).
//- No experience and no loot on the instance map.
//- Total : 2 rounds.
//===== Changelogs: ==========================================
//= 1.0 First version. [MondoTruth]
//= 1.1 iRO dialog and opponent scenes. [MondoTruth]
//============================================================

1@4igd,48,49,5	script(DISABLED)	Knight of Devotion#ig1	20825,{
	mes "[Knight of Devotion]";
	mes "Let's have a good fight.";
	close;
}

1@4igd,48,49,3	script(DISABLED)	Knight of Armament#ig1	20826,{
	mes "[Knight of Armament]";
	mes "Let's have a good fight.";
	close;
}

1@4igd,50,53,3	script	King's Knight#ig1	4_M_KNIGHT_SILVER,{
	.@devotion_first = (isbegin_quest(3438) > 0);
	switch('ig_step) {
	case 0:
		// A fresh hall always starts with round 1.
		'ig_devotion_first = .@devotion_first;
		for ( .@q = 3436; .@q <= 3439; .@q++ ) {
			if (isbegin_quest(.@q) > 0)
				erasequest .@q;
		}
		if (.@devotion_first)
			setquest 3438;
		else
			setquest 3436;
		mes "[King's Knight]";
		mes "Welcome to the Tourney Hall of Trials.";
		next;
		mes "[King's Knight]";
		mes "The rules of this tourney are simple.";
		next;
		mes "[King's Knight]";
		mes "You'll fight one-on-one, and whoever beats the other wins and moves onto the next round.";
		next;
		callsub S_ShowOpponent, .@devotion_first;
		mes "[King's Knight]";
		if (.@devotion_first)
			mes "Your first opponent is the ^4d4dffKnight of Devotion^000000.";
		else
			mes "Your first opponent is the ^4d4dffKnight of Armament^000000.";
		next;
		callsub S_Opponent, .@devotion_first, 1;
		mes "[King's Knight]";
		mes "Then, let's get started.";
		next;
		mes "[King's Knight]";
		mes "Good luck.";
		close2;
		'ig_step = 1;
		callsub S_Spawn, .@devotion_first;
		end;
	case 1:
	case 3:
		mes "[King's Knight]";
		mes "Let's resume your match.";
		next;
		mes "[King's Knight]";
		mes "Focus on the match.";
		close;
	case 2:
		if (.@devotion_first) {
			completequest 3438;
			setquest 3439;
		} else {
			completequest 3436;
			setquest 3437;
		}
		mes "[King's Knight]";
		mes "That was a good fight.";
		next;
		callsub S_ShowOpponent, !.@devotion_first;
		mes "[King's Knight]";
		if (.@devotion_first)
			mes "Your next opponent is the ^4d4dffKnight of Armament^000000.";
		else
			mes "Your next opponent is the ^4d4dffKnight of Devotion^000000.";
		next;
		callsub S_Opponent, !.@devotion_first, 2;
		mes "[King's Knight]";
		mes "Whoever wins this match will earn the title of Imperial Guard.";
		next;
		mes "[King's Knight]";
		mes "Okay, then do your best.";
		close2;
		'ig_step = 3;
		callsub S_Spawn, !.@devotion_first;
		end;
	case 4:
		mes "[King's Knight]";
		mes "That was a good fight.";
		next;
		mes "[King's Knight]";
		mes "Good job proving your sense of justice.";
		next;
		mes "[King's Knight]";
		mes "The definition of justice tends to change, depending on the time you live in.";
		next;
		mes "[King's Knight]";
		mes "For now, I suggest you go with the flow. Later, maybe you can help prevent the same conflict from happening.";
		next;
		mes "[King's Knight]";
		mes "Congratulations on your winning. Would you like to accept the title of Imperial Guard now?";
		next;
		if (select( "Yes.", "I need to get ready first." ) == 2) {
			mes "[King's Knight]";
			mes "Please speak to me again after you are ready.";
			close;
		}
		if (Class != JOB_ROYAL_GUARD && Class != JOB_ROYAL_GUARD_T) {
			mes "[King's Knight]";
			mes "Hm? You're not a Royal Guard anymore. I can't give you the title like this.";
			close;
		}
		if (checkriding() || ismounting()) {
			mes "[King's Knight]";
			mes "Get down from your mount first.";
			close;
		}
		if (SkillPoint > 0) {
			mes "[King's Knight]";
			mes "It appears you still have skill points remaining.";
			next;
			mes "[King's Knight]";
			mes "Please try again after using them all.";
			close;
		}
		if (checkweight(490087,1) == 0) {
			mes "^ff0000Lighten your load before joining the tourney.^000000";
			close;
		}
		if (.@devotion_first)
			completequest 3439;
		else
			completequest 3437;
		completequest 3440;
		jobchange JOB_IMPERIAL_GUARD;
		getitem 490087,1;	// 4th_Q_Necklace
		specialeffect2 EF_JOBLVUP;
		mes "[King's Knight]";
		mes "Now, you're an Imperial Guard.";
		next;
		mes "[King's Knight]";
		mes "Carry yourself well to match your title,";
		next;
		mes "[King's Knight]";
		mes "and keep up the good work raising the king's honor.";
		close2;
		.@id = instance_id();
		warp "prt_cas",179,12;
		instance_destroy .@id;
		end;
	}
	end;

// Show the opponent before their dialog and before the next-round prompt.
// getarg(0): 1 = Knight of Devotion, 0 = Knight of Armament
S_ShowOpponent:
	if (getarg(0)) {
		disablenpc instance_npcname("Knight of Armament#ig1");
		enablenpc instance_npcname("Knight of Devotion#ig1");
	} else {
		disablenpc instance_npcname("Knight of Devotion#ig1");
		enablenpc instance_npcname("Knight of Armament#ig1");
	}
	return;

// What the opponent says before a round.
// getarg(0): 1 = Knight of Devotion, 0 = Knight of Armament
// getarg(1): the round
S_Opponent:
	if (getarg(0)) {
		mes "[Knight of Devotion]";
		if (getarg(1) == 1) {
			mes "We met again, just as I expected.";
			next;
			mes "[Knight of Devotion]";
			mes "I can't wait to have this match with a kindred spirit.";
		} else {
			mes "We meet here.";
			next;
			mes "[Knight of Devotion]";
			mes "I will not lose, for the sake of my justice.";
		}
	} else {
		mes "[Knight of Armament]";
		if (getarg(1) == 1) {
			mes "We meet again, just as I expected.";
			next;
			mes "[Knight of Armament]";
			mes "I hope this becomes a splendid contest between kindred spirits.";
		} else {
			mes "We meet here.";
			next;
			mes "[Knight of Armament]";
			mes "I'll show you the power of superior armament.";
		}
	}
	next;
	return;

// getarg(0): 1 = Knight of Devotion, 0 = Knight of Armament
S_Spawn:
	killmonster 'ig_map$, instance_npcname("King's Knight#ig1") + "::OnKnightDead";
	disablenpc instance_npcname("Knight of Devotion#ig1");
	disablenpc instance_npcname("Knight of Armament#ig1");
	if (getarg(0))
		monster 'ig_map$,48,49,"--en--","4JOB_DEVOTION_NIGHT",1, instance_npcname("King's Knight#ig1") + "::OnKnightDead";
	else
		monster 'ig_map$,48,49,"--en--","4JOB_ARMED_NIGHT",1, instance_npcname("King's Knight#ig1") + "::OnKnightDead";
	return;

OnKnightDead:
	if ('ig_step == 1) {
		'ig_step = 2;
		// The next opponent is already waiting when the player talks to the King.
		callsub S_ShowOpponent, !'ig_devotion_first;
	} else if ('ig_step == 3) {
		'ig_step = 4;
	}
	end;

OnInstanceInit:
	'ig_map$ = instance_mapname("1@4igd");
	'ig_step = 0;
	'ig_devotion_first = 0;
	disablenpc instance_npcname("Knight of Devotion#ig1");
	disablenpc instance_npcname("Knight of Armament#ig1");
	setmapflag 'ig_map$, MF_NOBASEEXP;
	setmapflag 'ig_map$, MF_NOJOBEXP;
	setmapflag 'ig_map$, MF_NOMOBLOOT;
	setmapflag 'ig_map$, MF_NOMVPLOOT;
	end;
}
