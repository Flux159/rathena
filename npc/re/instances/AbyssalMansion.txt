1@4sac,79,49,0	script(DISABLED)	#ac_tree	4_ENERGY_BLACK,{
	if ('ac_step == 0) {
		mes "- This spot looks good, but you should talk to Doomk first. -";
		close;
	}
	if ('ac_step > 1) {
		mes "- The darkness under the tree is thick and cold. -";
		close;
	}
	if (job_ac_4th == 7 && (countitem(6089) < 20 || countitem(7799) < 5)) {
		mes "- To call the dark elemental you need: -";
		mes "^0000ff20 Dark Piece^000000 (" + countitem(6089) + " / 20)";
		mes "^0000ff5 Crystal of Darkness^000000 (" + countitem(7799) + " / 5)";
		close;
	}
	unittalk getcharid(3), strcharinfo(0) + " : This spot looks good.";
	mes "- The Crystals of Darkness and Dark Pieces inside my pocket reacted. -";
	if (job_ac_4th == 7) {
		delitem 6089,20;	// Piece_Of_Darkness
		delitem 7799,5;	// Dark_Crystal
		job_ac_4th = 8;
	}
	'ac_step = 2;
	close2;
	disablenpc instance_npcname("#ac_tree");
	getmapxy(.@map$, .@x, .@y, BL_PC);
	.@npc$[0] = instance_npcname("Doomk#ac1");
	.@npc$[1] = instance_npcname("Mayssel#ac1");
	.@npc$[2] = instance_npcname("Verkhasel#ac1");
	setarray .@dx[0], 4, 5, 1;
	setarray .@dy[0], 0, -1, 2;
	setarray .@fx[0], 83, 84, 80;
	setarray .@fy[0], 49, 48, 51;
	for (.@i = 0; .@i < 3; .@i++) {
		if (checkcell(.@map$, .@x + .@dx[.@i], .@y + .@dy[.@i], cell_chkpass))
			movenpc .@npc$[.@i], .@x + .@dx[.@i], .@y + .@dy[.@i];
		else
			movenpc .@npc$[.@i], .@fx[.@i], .@fy[.@i];
		enablenpc .@npc$[.@i];
		specialeffect EF_TELEPORTATION, AREA, .@npc$[.@i];
	}
	enablenpc instance_npcname("Blackie#ac1");
	specialeffect EF_DARKBREATH, AREA, instance_npcname("Blackie#ac1");
	sleep2 500;
	npctalk "Verkhasel : What's this energy?", .@npc$[2];
	sleep2 700;
	npctalk "Blackie : I'm back!", instance_npcname("Blackie#ac1");
	sleep2 700;
	npctalk "Mayssel : What? What's going on?", .@npc$[1];
	end;
}

1@4sac,1,1,0	script	#ac_fight	-1,{
	end;

OnStart:
	if ('ac_step != 3)
		end;
	.@verkhasel$ = instance_npcname("Verkhasel#ac1");
	getmapxy(.@map$, .@x, .@y, BL_NPC, .@verkhasel$);
	specialeffect EF_DARKBREATH, AREA, .@verkhasel$;
	disablenpc .@verkhasel$;
	monster 'ac_map$, .@x, .@y, "Possessed Verkhasel", "4JOB_VERKHASEL", 1, instance_npcname("#ac_fight") + "::OnVerkhaselDead";
	'ac_mob = $@mobid[0];
	initnpctimer;
	end;

OnTimer1000:
	if (unitexists('ac_mob))
		unittalk 'ac_mob, "Possessed Verkhasel : Wait... Wha...?";
	end;

OnTimer3000:
	npctalk "Blackie : Now, fight! Fight and win!", instance_npcname("Blackie#ac1");
	stopnpctimer;
	end;

OnVerkhaselDead:
	if ('ac_step != 3)
		end;
	stopnpctimer;
	'ac_step = 4;
	.@verkhasel$ = instance_npcname("Verkhasel#ac1");
	enablenpc .@verkhasel$;
	npctalk "Verkhasel : Grr, you used me!", .@verkhasel$;
	npctalk "Blackie : I like you. You're more compatible with me than I thought!", instance_npcname("Blackie#ac1");
	mapannounce 'ac_map$, "Blackie lets go of Verkhasel. Talk to Blackie.", bc_map, 0xFFFF00;
	end;
