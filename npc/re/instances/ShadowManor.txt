1@4sac,1,1,0	script	#sc_fight	-1,{
	end;

OnStart:
	if ('sc_step != 3)
		end;
	.@doomk$ = instance_npcname("Doomk#sc1");
	getmapxy(.@map$, .@x, .@y, BL_NPC, .@doomk$);
	specialeffect EF_DARKBREATH, AREA, .@doomk$;
	disablenpc .@doomk$;
	monster 'sc_map$, .@x, .@y, "Possessed Doomk", "4JOB_DOOMK", 1, instance_npcname("#sc_fight") + "::OnDoomkDead";
	'sc_mob = $@mobid[0];
	initnpctimer;
	end;

OnTimer1000:
	npctalk "Let's see what you've got.", instance_npcname("Blackie#sc1");
	end;

OnTimer3000:
	if (unitexists('sc_mob))
		unittalk 'sc_mob, "You... You won't get away with this!";
	end;

OnTimer5000:
	npctalk "Now, fight! Fight and win!", instance_npcname("Blackie#sc1");
	end;

OnTimer8000:
	npctalk "Good luck, Doomk.", instance_npcname("Verkhasel#sc1");
	stopnpctimer;
	end;

OnDoomkDead:
	if ('sc_step != 3)
		end;
	stopnpctimer;
	'sc_step = 4;
	.@doomk$ = instance_npcname("Doomk#sc1");
	enablenpc .@doomk$;
	npctalk "Grr...", .@doomk$;
	npctalk "I like you. You're more compatible with me than I thought!", instance_npcname("Blackie#sc1");
	end;
}
