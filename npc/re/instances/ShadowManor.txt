//===== Description: =========================================
//- [Walkthrough conversion]
//- Part of the Shadow Cross job change (npc/re/jobs/4-1/shadow_cross.txt).
//- Follows the official quest flow as seen in iRO and kRO
//- gameplay videos; the dialog is our own wording, not a
//- translation of the official text.
//- Verkhasel, Mayssel and Doomk come in with you. Mayssel says
//- the Shadow likes dark places such as a big garden tree; Doomk
//- and Mayssel go searching on their own. Under the tree the
//- materials react, all three come back and Blackie appears.
//- Blackie tests you: it takes Doomk over and you fight him.
//- Talked to again after the fight, Blackie gives you its power;
//- Mayssel and Doomk congratulate you.
//- Unknown, picked by us:
//-  - the cells of every NPC and of the tree (no map of the
//-    official spots); the tree spot is next to the big tree
//-    block at 68-72,72-77,
//-  - the fight: the iRO video shows a monster labelled "Shadow"
//-    and then "Doomk" (battle log: "Possessed Doomk" and
//-    "Someone"); here it is one monster, the fork's 4JOB_DOOMK,
//-    named "Possessed Doomk",
//-  - who says "What's this energy?", "What? What's going on?"
//-    and "You won't get away with this!" (the chat log has no
//-    names for them),
//-  - when the materials are used: kRO's quest text says they
//-    are for summoning the spirit, so they go at the tree,
//-  - Doomk's sprite (4_M_SHADOWCHASER), 4JOB_DOOMK's stats and
//-    the effects of the appearances.
//- Quest log (kRO): 18065 on entering, 18066 once Blackie is
//- summoned, 18067 once it offered you its power. Giving up has
//- no waiting time: kRO's text only says you can go in again.
//- No experience and no loot on the instance map.
//===== Changelogs: ==========================================
//= 1.0 First version. [MondoTruth]
//= 1.1 Allies Mayssel and Doomk, the garden tree, Blackie's
//=     test, as in the videos. [MondoTruth]
//= 1.2 kRO's quest ids; no waiting time after giving up.
//=     [MondoTruth]
//= 1.3 Scenes as in the iRO video: Doomk and Mayssel leave,
//=     come back at the tree and after the job change; the
//=     "Shadow" fight with overhead lines; cutins and sprites.
//=     [MondoTruth]
//= 1.4 Blackie does the job change. [MondoTruth]
//= 1.5 Every dialog page and overhead line checked against the
//=     iRO video again: all three come back at the tree, Doomk
//=     is the one you fight. [MondoTruth]
//============================================================

// 'sc_step:
//  0 just entered
//  1 Mayssel's hint: find a dark place, a big garden tree
//  2 Blackie was called at the tree
//  3 Blackie took over Doomk: fight
//  4 Doomk is down, talk to Blackie
//  5 Blackie offers its power: talk to it again
//  6 Shadow Cross: Verkhasel takes you out

1@4sac,91,25,3	script	Verkhasel#sc1	4_M_GUILLOTINE,{
	cutin "gc_verkhasel01.bmp",0;
	switch('sc_step) {
	case 0:
		mes "[Verkhasel]";
		mes "Remember this place? We came here on our last operation together.";
		mes "You became a Guillotine Cross right after that.";
		next;
		cutin "3rd_sc_doomk01.bmp",2;
		mes "[Doomk]";
		mes "Huh, I never knew this place existed.";
		mes "We haven't met, have we? I'm Doomk, I lead the Shadow Chasers.";
		next;
		mes "[Doomk]";
		mes "Right, let's go and hunt down this dark elemental, or whatever it is.";
		next;
		cutin "gc_mayssel01.bmp",1;
		mes "[Mayssel]";
		mes "We're not hunting it, Doomk. We're calling it.";
		mes "As blunt as ever, I see.";
		next;
		cutin "gc_verkhasel01.bmp",0;
		mes "[Verkhasel]";
		mes "Mayssel, where do we go from here?";
		mes "You're the only one of us who has taken in its power so far.";
		next;
		cutin "gc_mayssel01.bmp",1;
		mes "[Mayssel]";
		mes "Even I don't know everything about it.";
		mes "For a start, let's find a place the dark elemental would like.";
		next;
		mes "[Mayssel]";
		mes "It likes places that stay dark and shady even at midday, where it can hide from the light.";
		mes "Under a ^0000ffbig garden tree^000000, for instance.";
		next;
		cutin "3rd_sc_doomk01.bmp",2;
		mes "[Doomk]";
		mes "Then why don't we split up and look around?";
		mes "It'll answer wherever it likes, won't it?";
		next;
		specialeffect EF_HIDING, AREA, instance_npcname("Doomk#sc1");
		disablenpc instance_npcname("Doomk#sc1");
		cutin "gc_mayssel01.bmp",1;
		mes "[Mayssel]";
		mes "Fine, I'll do the same.";
		mes "To be honest, I doubt it cares where we call it from.";
		next;
		disablenpc instance_npcname("Mayssel#sc1");
		cutin "gc_verkhasel01.bmp",0;
		mes "[Verkhasel]";
		mes "Not exactly team players, those two.";
		mes "Look around and find the spot where the dark elemental reacts.";
		'sc_step = 1;
		break;
	case 1:
	case 2:
	case 3:
		mes "[Verkhasel]";
		if ('sc_step == 1)
			mes "Find a big tree in the garden, somewhere that stays dark.";
		else
			mes "Go on. It wants to see what you can do.";
		next;
		if (select( "All right.", "Give up." ) == 2) {
			mes "[Verkhasel]";
			mes "Giving up ends this. You can come back and try again later.";
			mes "Are you sure?";
			next;
			if (select( "Stay.", "Leave." ) == 2) {
				cutin "",255;
				.@id = instance_id();
				warp "ve_fild05",317,232;
				instance_destroy .@id;
				end;
			}
		}
		break;
	case 4:
	case 5:
		mes "[Verkhasel]";
		mes "It's waiting for you. Go and talk to it.";
		break;
	case 6:
		mes "[Verkhasel]";
		mes "Let's get out of here. Ready to go back?";
		next;
		if (select( "Not yet.", "Let's go." ) == 2) {
			cutin "",255;
			.@id = instance_id();
			warp "ve_fild05",317,232;
			instance_destroy .@id;
			end;
		}
		break;
	}
	close2;
	cutin "",255;
	end;

OnInstanceInit:
	'sc_map$ = instance_mapname("1@4sac");
	'sc_step = 0;
	setmapflag 'sc_map$, MF_NOBASEEXP;
	setmapflag 'sc_map$, MF_NOJOBEXP;
	setmapflag 'sc_map$, MF_NOMOBLOOT;
	setmapflag 'sc_map$, MF_NOMVPLOOT;
	end;
}

// The spot under the big tree where the dark elemental reacts.
1@4sac,70,71,0	script	#sc_tree	4_ENERGY_BLACK,{
	if ('sc_step == 0) {
		mes "- A cold, dark spot under a big tree. -";
		mes "- Better talk to Verkhasel first. -";
		close;
	}
	if ('sc_step > 1) {
		mes "- The darkness under the tree is thick and cold. -";
		close;
	}
	if (job_sc_4th == 7 && (countitem(7799) < 5 || countitem(6089) < 20)) {
		mes "- A cold, dark spot under a big tree. -";
		mes "- To call the Shadow you need: -";
		mes "^0000ff5 Crystal of Darkness^000000 (" + countitem(7799) + " / 5)";
		mes "^0000ff20 Dark Piece^000000 (" + countitem(6089) + " / 20)";
		close;
	}
	unittalk getcharid(3), strcharinfo(0) + " : This looks like a good spot.";
	mes "- The Crystals of Darkness and the Dark Pieces in your bag stir. -";
	mes "- They're reacting to something here. -";
	if (job_sc_4th == 7) {
		delitem 7799,5;	// Dark_Crystal
		delitem 6089,20;	// Piece_Of_Darkness
		job_sc_4th = 8;
		changequest 18065,18066;
	}
	'sc_step = 2;
	close2;
	// Verkhasel, Mayssel and Doomk join you; Blackie shows itself.
	getmapxy(.@map$, .@x, .@y, BL_PC);
	setarray .@npc$[0], instance_npcname("Verkhasel#sc1"), instance_npcname("Mayssel#sc1"), instance_npcname("Doomk#sc1");
	setarray .@dx[0], 2, 1, -2;
	setarray .@dy[0], 0, -2, 0;
	setarray .@fx[0], 73, 75, 67;
	setarray .@fy[0], 69, 70, 70;
	for (.@i = 0; .@i < 3; .@i++) {
		disablenpc .@npc$[.@i];
		if (checkcell(.@map$, .@x + .@dx[.@i], .@y + .@dy[.@i], cell_chkpass))
			movenpc .@npc$[.@i], .@x + .@dx[.@i], .@y + .@dy[.@i];
		else
			movenpc .@npc$[.@i], .@fx[.@i], .@fy[.@i];
		enablenpc .@npc$[.@i];
		specialeffect EF_TELEPORTATION, AREA, .@npc$[.@i];
	}
	sleep2 500;
	npctalk "Doomk : I'm back!", .@npc$[2];
	sleep2 1000;
	npctalk "Verkhasel : What's this energy?", .@npc$[0];
	sleep2 1000;
	npctalk "Mayssel : What? What's happening?", .@npc$[1];
	sleep2 500;
	enablenpc instance_npcname("Blackie#sc1");
	specialeffect EF_DARKBREATH, AREA, instance_npcname("Blackie#sc1");
	end;
}

1@4sac,1,1,0	script	#sc_fight	-1,{
	end;

OnStart:
	if ('sc_step != 3)
		end;
	// Blackie takes Doomk over: the NPC goes, the monster comes in his place.
	.@doomk$ = instance_npcname("Doomk#sc1");
	getmapxy(.@map$, .@x, .@y, BL_NPC, .@doomk$);
	specialeffect EF_DARKBREATH, AREA, .@doomk$;
	disablenpc .@doomk$;
	monster 'sc_map$, .@x, .@y, "Possessed Doomk", "4JOB_DOOMK", 1, instance_npcname("#sc_fight") + "::OnDoomkDead";
	'sc_mob = $@mobid[0];
	initnpctimer;
	end;

OnTimer1000:
	npctalk "Blackie : Let's see what you've got.", instance_npcname("Blackie#sc1");
	end;

OnTimer3000:
	if (unitexists('sc_mob))
		unittalk 'sc_mob, "Possessed Doomk : You... You won't get away with this!";
	end;

OnTimer5000:
	npctalk "Blackie : Now fight! Fight and win!", instance_npcname("Blackie#sc1");
	end;

OnTimer8000:
	npctalk "Verkhasel : Good luck, Doomk.", instance_npcname("Verkhasel#sc1");
	end;

OnTimer14000:
	npctalk "Blackie : I like you. You suit me better than I thought!", instance_npcname("Blackie#sc1");
	stopnpctimer;
	end;

OnDoomkDead:
	if ('sc_step != 3)
		end;
	stopnpctimer;
	'sc_step = 4;
	// Doomk is himself again where he fell.
	.@doomk$ = instance_npcname("Doomk#sc1");
	enablenpc .@doomk$;
	npctalk "Doomk : Grr, you used me!", .@doomk$;
	mapannounce 'sc_map$, "Blackie lets go of Doomk. Talk to Blackie.", bc_map, 0xFFFF00;
	end;
}
