	mes "[Verkhasel]";
	mes "Here we are. Good work getting the materials for the summoning. That should be plenty.";
	mes "Mayssel will catch up soon. Let's go on ahead.";
	next;
	mes "[Verkhasel]";
	mes "There's no promise this dark elemental will agree to our terms when it appears.";
	mes "Let's be ready for anything before we call it.";
	next;
	mes "[Verkhasel]";
	mes "Tell me when you're ready to go in.";
	next;
	switch( select( "Open the way to the Shadow Manor.", "Enter the Shadow Manor." ) ) {
	case 1:
		.@id = instance_create(.@md_name$);
		mes "[Verkhasel]";
		if (.@id == -3)
			mes "The way is already open. Choose to enter.";
		else if (.@id < 0)
			mes "The way won't open right now. Try again in a moment.";
		else
			mes "There. Talk to me again when you're ready to go in.";
		break;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_OTHER:
			mes "^ff0000An unknown error occurred.^000000";
			break;
		case IE_NOINSTANCE:
			mes "[Verkhasel]";
			mes "The way isn't open yet. Ask me to open it first.";
			break;
		case IE_NOMEMBER:
			mes "[Verkhasel]";
			mes "Only the party that opened the way may enter.";
			break;
		case IE_OK:
			cutin "",255;
			if (isbegin_quest(18064) == 1)
				changequest 18064,18065;
			mapannounce "ve_fild05", strcharinfo(0) + " of the " + getpartyname( getcharid(1) ) + " party is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			end;
		}
		break;
	}
	close2;
	cutin "",255;
	end;
