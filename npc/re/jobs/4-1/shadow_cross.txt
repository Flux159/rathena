ve_fild05,317,235,3	script	Verkhasel#sc_start	467,{
	.@md_name$ = "Shadow Manor";

	if (job_sc_4th < 7 || (Class != JOB_GUILLOTINE_CROSS && Class != JOB_GUILLOTINE_CROSS_T)) {
		mes "- Someone is watching the road from the shadows. -";
		close;
	}
	// After the summoning (job_sc_4th 8) the materials are gone; only the fight is left.
	if (job_sc_4th == 7 && (countitem(7799) < 5 || countitem(6089) < 20)) {
		mes "[Verkhasel]";
		mes "We can't call the Shadow without the materials. Bring me:";
		mes "^0000ff5 Crystal of Darkness^000000 (" + countitem(7799) + " / 5)";
		mes "^0000ff20 Dark Piece^000000 (" + countitem(6089) + " / 20)";
		close;
	}
	if (SkillPoint > 0) {
		mes "[Verkhasel]";
		mes "You still have skill points left. Learn your skills first.";
		close;
	}
	if (getcharid(1) == 0 || is_party_leader() == false) {
		mes "[Verkhasel]";
		mes "Make a party of your own and lead it.";
		close;
	}
	getpartymember getcharid(1), 0;
	if ($@partymembercount > 1) {
		mes "[Verkhasel]";
		mes "This is your job. Make a party with ^0000ffonly yourself^000000 in it.";
		close;
	}
	mes "[Verkhasel]";
	mes "The mansion is ready. Mayssel and Doomk are coming too.";
	mes "Shall we go?";
	next;
	switch( select( "Enter the Shadow Manor.", "Open the way to the Shadow Manor." ) ) {
	case 2:
		.@id = instance_create(.@md_name$);
		if (.@id == -3) {
			mes "[Verkhasel]";
			mes "The way is already open. Choose to enter.";
			close;
		}
		if (.@id < 0) {
			mes "[Verkhasel]";
			mes "The way won't open right now. Try again in a moment.";
			close;
		}
		mes "[Verkhasel]";
		mes "There. Talk to me again when you're ready to go in.";
		close;
	case 1:
		switch( instance_enter(.@md_name$) ) {
		case IE_OTHER:
			mes "^ff0000An unknown error occurred.^000000";
			close;
		case IE_NOINSTANCE:
			mes "[Verkhasel]";
			mes "The way isn't open yet. Ask me to open it first.";
			close;
		case IE_NOMEMBER:
			mes "[Verkhasel]";
			mes "Only the party that opened the way may enter.";
			close;
		case IE_OK:
			if (isbegin_quest(18064) == 1)
				changequest 18064,18065;
			mapannounce "ve_fild05", strcharinfo(0) + " of the " + getpartyname( getcharid(1) ) + " party is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			end;
		}
		end;
	}
	end;
}
