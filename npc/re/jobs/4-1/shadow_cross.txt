//===== Description: =========================================
//- [Walkthrough conversion]
//- Follows the official quest flow as seen in iRO and kRO
//- gameplay videos; the dialog is our own wording, not a
//- translation of the official text.
//- The last part takes place in an instance
//- (npc/re/instances/ShadowManor.txt).
//- Field positions from the iRO Wiki guide and the videos:
//- the pale shadow (ve_fild07 ~101,141) and Verkhasel north
//- of Veins (ve_fild05 ~318,235) were read off the minimap.
//- Sprites and cutins as in the iRO video: Rumin 4_M_AC_RUMIN

//- 4_F_MAYSEL (gc_mayssel), Vicente 4_M_HUMAN_02 (sc_vicente).
//- Mayssel is only in the safe house while she briefs you.
//- Unknown, picked by us:
//-  - the sprites of Noah, Tori, the mother and Blackie,
//-  - the cells of Mayssel, Vicente, the mother, Noah and Tori,
//-  - who sends you to Noah (here: Tori's mother).
//- Quest log: kRO's chain 18059 "Shadow's Call" and 18060-18067
//- "Job Change to Shadow Cross-1" to "-8" (the client's quest
//- table); each step below moves to the next one. kRO has no
//- waiting time after giving up in the mansion.
//- The videos name the shadow "Blackie"; Noah once calls it
//- "Noir" in iRO, probably a translation slip, so we use Blackie.
//===== Changelogs: ==========================================
//= 1.0 First version. [MondoTruth]
//= 1.1 Story follows the videos: Mayssel and Vicente, the
//=     bead, the talking Blackie, the garden tree. [MondoTruth]
//= 1.2 kRO's quest ids 18059-18067, one per step; no waiting
//=     time after giving up. [MondoTruth]
//= 1.3 Sprites, cutins and overhead lines as in the iRO video;
//=     Mayssel appears for the briefing only; you greet Blackie
//=     out loud instead of typing its name; the way to the
//=     mansion is marked for the navigation. [MondoTruth]
//============================================================

ve_fild05,317,235,3	script	Verkhasel#sc_start	4_M_GUILLOTINE,{
	.@md_name$ = "Shadow Manor";

	if (job_sc_4th < 7 || (Class != JOB_GUILLOTINE_CROSS && Class != JOB_GUILLOTINE_CROSS_T)) {
		mes "- Someone is watching the road from the shadows. -";
		close;
	}
	cutin "gc_verkhasel01.bmp",0;
	// After the summoning (job_sc_4th 8) the materials are gone; only the fight is left.
	if (job_sc_4th == 7 && (countitem(7799) < 5 || countitem(6089) < 20)) {
		mes "[Verkhasel]";
		mes "We can't call the Shadow without the materials. Bring me:";
		mes "^0000ff5 Crystal of Darkness^000000 (" + countitem(7799) + " / 5)";
		mes "^0000ff20 Dark Piece^000000 (" + countitem(6089) + " / 20)";
		close2;
		cutin "",255;
		end;
	}
	if (SkillPoint > 0) {
		mes "[Verkhasel]";
		mes "You still have skill points left. Learn your skills first.";
		close2;
		cutin "",255;
		end;
	}
	if (getcharid(1) == 0 || is_party_leader() == false) {
		mes "[Verkhasel]";
		mes "There's no telling whether it will accept our terms, so let's be ready for anything.";
		mes "Make a party of your own and lead it, then talk to me.";
		close2;
		cutin "",255;
		end;
	}
	getpartymember getcharid(1), 0;
	if ($@partymembercount > 1) {
		mes "[Verkhasel]";
		mes "This is your job. Make a party with ^0000ffonly yourself^000000 in it.";
		close2;
		cutin "",255;
		end;
	}
	mes "[Verkhasel]";
	mes "There's no telling whether it will accept our terms, so let's be ready for anything.";
	mes "The mansion is ready. Mayssel and Doomk are coming too.";
	mes "Shall we go?";
	next;
	switch( select( "Enter the Shadow Manor.", "Open the way to the Shadow Manor." ) ) {
	case 2:
		.@id = instance_create(.@md_name$);
		mes "[Verkhasel]";
		if (.@id == -3)
			mes "The way is already open. Choose to enter.";
		else if (.@id < 0)
			mes "The way won't open right now. Try again in a moment.";
		else
			mes "There. Talk to me again when you're ready to go in.";
		break;
	case 1:
		switch( instance_enter(.@md_name$) ) {
		case IE_OTHER:
			mes "^ff0000An unknown error occurred.^000000";
			break;
		case IE_NOINSTANCE:
			mes "[Verkhasel]";
			mes "The way isn't open yet. Ask me to open it first.";
			break;
		case IE_NOMEMBER:
			mes "[Verkhasel]";
			mes "Only the party that opened the way may enter.";
			break;
		case IE_OK:
			cutin "",255;
			if (isbegin_quest(18064) == 1)
				changequest 18064,18065;
			mapannounce "ve_fild05", strcharinfo(0) + " of the " + getpartyname( getcharid(1) ) + " party is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			end;
		}
		break;
	}
	close2;
	cutin "",255;
	end;
}
