ve_fild05,313,110,4	script	Doomk#ac_start	4_M_SHADOWCHASER,{
	.@md_name$ = "Abyssal Mansion";
	if (job_ac_4th < 6 || (Class != JOB_SHADOW_CHASER && Class != JOB_SHADOW_CHASER_T)) {
		mes "[Doomk]";
		mes "This is no place for an idle visit, adventurer.";
		close;
	}
	cutin "3rd_sc_doomk01.bmp",2;
	if (job_ac_4th == 6 && (countitem(6089) < 20 || countitem(7799) < 5)) {
		mes "[Doomk]";
		mes "We need the dark objects Vicente asked for:";
		mes "^0000ff20 Dark Piece^000000 (" + countitem(6089) + " / 20)";
		mes "^0000ff5 Crystal of Darkness^000000 (" + countitem(7799) + " / 5)";
		close2;
		cutin "",255;
		end;
	}
	if (SkillPoint > 0) {
		mes "[Doomk]";
		mes "You still have skill points left. Learn your skills first.";
		close2;
		cutin "",255;
		end;
	}
	if (getcharid(1) == 0 || is_party_leader() == false) {
		mes "[Doomk]";
		mes "Make a party of your own and lead it, then talk to me.";
		close2;
		cutin "",255;
		end;
	}
	getpartymember getcharid(1), 0;
	if ($@partymembercount > 1) {
		mes "[Doomk]";
		mes "This is your test. Make a party with ^0000ffonly yourself^000000 in it.";
		close2;
		cutin "",255;
		end;
	}
	mes "[Doomk]";
	mes "I've been waiting since I got word from";
	mes "Vincente.";
	mes "Good job bringing Crystals of";
	mes "Darkness and Dark Pieces.";
	next;
	mes "[Doomk]";
	mes "Verkhasel will open the gate when";
	mes "you're ready.";
	mes "There's no guarantee the dark";
	mes "elemental will do as we say. You'd";
	mes "better be prepared for a fight.";
	next;
	mes "[Doomk]";
	mes "I trust you, but if you cant trust yourself,";
	mes "then I'll give you some time to be";
	mes "prepared.";
	next;
	switch(select("Apply for the Abyssal Mansion.", "Enter the Abyssal Mansion.")) {
	case 1:
		.@id = instance_create(.@md_name$);
		mes "[Doomk]";
		if (.@id == -3)
			mes "The gate is already open. Choose to enter.";
		else if (.@id < 0)
			mes "The gate won't open right now. Try again in a moment.";
		else
			mes "There. Talk to me again when you're ready to go in.";
		break;
	case 2:
		switch(instance_enter(.@md_name$)) {
		case IE_OTHER:
			mes "^ff0000An unknown error occurred.^000000";
			break;
		case IE_NOINSTANCE:
			mes "[Doomk]";
			mes "The gate isn't open yet. Apply for the mansion first.";
			break;
		case IE_NOMEMBER:
			mes "[Doomk]";
			mes "Only the party that opened the gate may enter.";
			break;
		case IE_OK:
			cutin "",255;
			if (job_ac_4th == 6) {
				job_ac_4th = 7;
				changequest 18075,18076;
			}
			mapannounce "ve_fild05", strcharinfo(0) + " of the " + getpartyname(getcharid(1)) + " party is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			end;
		}
		break;
	}
	close2;
	cutin "",255;
	end;

OnInit:
	questinfo(QTYPE_JOB, QMARK_YELLOW, "(Class == JOB_SHADOW_CHASER || Class == JOB_SHADOW_CHASER_T) && job_ac_4th >= 6");
	end;
