prt_cas,181,10,3	script	King's Knight#ig_start	4_M_KNIGHT_SILVER,{
	.@md_name$ = "Tourney Hall of Trials";

	if (Class == JOB_IMPERIAL_GUARD) {
		mes "[King's Knight]";
		mes "You're the winner of the final match.";
		next;
		mes "[King's Knight]";
		mes "Congratulations on your earning such an honorable title. Please use your power to raise your honor.";
		close;
	}
	if (Class != JOB_ROYAL_GUARD && Class != JOB_ROYAL_GUARD_T) {
		mes "[King's Knight]";
		mes "I watch over the king's guardians.";
		mes "Only Royal Guards may take part in the Palace's tourney.";
		close;
	}
	if (BaseLevel < 200 || JobLevel < 70) {
		mes "[King's Knight]";
		mes "You serve the king well, Royal Guard, but you are not ready for the tourney yet.";
		mes "Come back at ^dd5555Base Level 200 and Job Level 70^000000.";
		close;
	}
	if (isbegin_quest(3430) == 0 && isbegin_quest(3433) == 0 && isbegin_quest(3432) == 0
		&& isbegin_quest(3440) == 0) {
		mes "[King's Knight]";
		mes "The Royal Guards exist for the king.";
		mes "They're considered the best of all knights because of their faith in their god and loyalty toward their king.";
		next;
		mes "[King's Knight]";
		mes "Over time, though, some Royal Guards crumble under the pressure of constant threats and hardship and forget who they are.";
		next;
		mes "[King's Knight]";
		mes "That's why the group needs culling every once in a while.";
		next;
		mes "[King's Knight]";
		mes "It only needs those who can stay as the honorable royal knights they are, fighting for their power and justice.";
		next;
		mes "[King's Knight]";
		mes "For that, the Palace has decided to host a tourney.";
		next;
		mes "[King's Knight]";
		mes "Are you using your power for the right reasons?";
		next;
		mes "[King's Knight]";
		mes "Do you have an unshakable belief in yourself?";
		next;
		mes "[King's Knight]";
		mes "Then, how'd you like to join this tourney?";
		next;
		if (select( "I'd love to.", "Let me think..." ) == 2) {
			mes "[King's Knight]";
			mes "Hesitation is not a good emotion.";
			next;
			mes "[King's Knight]";
			mes "Make a swift decision and come back to me.";
			close;
		}
		setquest 3430;
		setquest 3431;
		mes "[King's Knight]";
		mes "I said the group needs culling. Remember?";
		next;
		mes "[King's Knight]";
		mes "You should see it for yourself.";
		next;
		mes "[King's Knight]";
		mes "Before joining the tourney, meet and talk to these two knights.";
		next;
		mes "[King's Knight]";
		mes "First, the ^4d4dffKnight of Devotion^000000.";
		next;
		mes "[King's Knight]";
		mes "She's on the <NAVI>^4D4DFF[northwest side of Prontera]^000000<INFO>prontera,58,364,000,1,</INFO></NAVI>.";
		next;
		mes "[King's Knight]";
		mes "The other one is the ^4d4dffKnight of Armament^000000.";
		next;
		mes "[King's Knight]";
		mes "He's on the <NAVI>^4D4DFF[northeast side of Prontera]^000000<INFO>prontera,276,355,000,1,</INFO></NAVI>.";
		next;
		mes "[King's Knight]";
		mes "Talk to them and think about which side you're on.";
		next;
		mes "[King's Knight]";
		mes "You can come back to me and apply for the tourney afterward.";
		next;
		mes "[King's Knight]";
		mes "Now, see you.";
		close;
	}
	if (isbegin_quest(3430) == 1 || isbegin_quest(3431) == 1) {
		mes "[King's Knight]";
		mes "Have you met both knights yet?";
		if (isbegin_quest(3430) == 1)
			mes "The ^4d4dffKnight of Devotion^000000 is in the <NAVI>^4D4DFF[northwest of Prontera]^000000<INFO>prontera,58,364,000,1,</INFO></NAVI>.";
		if (isbegin_quest(3431) == 1)
			mes "The ^4d4dffKnight of Armament^000000 is in the <NAVI>^4D4DFF[northeast of Prontera]^000000<INFO>prontera,276,355,000,1,</INFO></NAVI>.";
		close;
	}
	if (isbegin_quest(3440) == 0) {
		.@devotion_faith = (isbegin_quest(3433) == 1);
		.@armament_faith = (isbegin_quest(3435) == 1);
		if (.@devotion_faith != .@armament_faith) {
			mes "[King's Knight]";
			mes "Huh???";
			next;
			mes "[King's Knight]";
			mes "If we'd had a proper conversation, you wouldn't have answered like that.";
			next;
			mes "[King's Knight]";
			mes "Shouldn't you show some conviction in your opinion?";
			next;
			mes "[King's Knight]";
			mes "This won't do. You must return and discuss this further.";
			close2;
			for ( .@q = 3432; .@q <= 3435; .@q++ ) {
				if (isbegin_quest(.@q) > 0)
					erasequest .@q;
			}
			setquest 3430;
			setquest 3431;
			end;
		}
		.@faith = .@devotion_faith;
		mes "[King's Knight]";
		mes "I see.";
		next;
		mes "[King's Knight]";
		if (.@faith)
			mes "Your faith and loyalty, huh.";
		else
			mes "You've chosen powerful equipment and exceptional swordsmanship.";
		next;
		mes "[King's Knight]";
		mes "Okay, then show me what you can do with them.";
		next;
		mes "[King's Knight]";
		mes "Let me know when you're ready to join the tourney.";
		close2;
		for ( .@q = 3432; .@q <= 3435; .@q++ ) {
			if (isbegin_quest(.@q) > 0)
				erasequest .@q;
		}
		setquest 3440;
		job_ig_4th_answer = 0;
		// Round 1 is the knight who shares the player's view.
		if (.@faith)
			setquest 3438;
		else
			setquest 3436;
		end;
	}
	if (SkillPoint > 0) {
		mes "[King's Knight]";
		mes "You still have skill points left to learn.";
		mes "Use all of them before the tourney.";
		close;
	}
	if (getcharid(1) == 0 || is_party_leader() == false) {
		mes "[King's Knight]";
		mes "The tourney hall is opened for a party leader.";
		mes "Make a party of your own and lead it.";
		close;
	}
	getpartymember getcharid(1), 0;
	if ($@partymembercount > 1) {
		mes "[King's Knight]";
		mes "The rounds are fought one-on-one.";
		mes "Make a party with ^0000ffonly yourself^000000 in it.";
		close;
	}
	mes "[King's Knight]";
	mes "Do you want to apply for the tourney of trials?";
	next;
	switch( select( "Enter the Tourney Hall of Trials.", "Request Tourney Hall of Trials." ) ) {
	case 2:
		.@id = instance_create(.@md_name$);
		if (.@id == -3) {
			mes "[King's Knight]";
			mes "The tourney hall is already prepared for you.";
			mes "Choose to enter it.";
			close;
		}
		if (.@id < 0) {
			mes "[King's Knight]";
			mes "The hall can't be prepared right now. Try again in a moment.";
			close;
		}
		mes "[King's Knight]";
		mes "The tourney hall is being prepared.";
		mes "Talk to me again when you want to go in.";
		close;
	case 1:
		switch( instance_enter(.@md_name$) ) {
		case IE_OTHER:
			mes "^ff0000An unknown error occurred.^000000";
			close;
		case IE_NOINSTANCE:
			mes "[King's Knight]";
			mes "The tourney hall isn't ready yet. Request it first.";
			close;
		case IE_NOMEMBER:
			mes "[King's Knight]";
			mes "Only the party that requested the hall may enter.";
			close;
		case IE_OK:
			mapannounce "prt_cas", strcharinfo(0) + " of the " + getpartyname( getcharid(1) ) + " party is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			end;
		}
		end;
	}
	end;

OnInit:
	questinfo(QTYPE_JOB, QMARK_YELLOW, "(Class == JOB_ROYAL_GUARD || Class == JOB_ROYAL_GUARD_T) && BaseLevel >= 200 && JobLevel >= 70");
	end;
}

	mes "But I was gifted in sword-fighting.";
	next;
	mes "[Knight of Devotion]";
	mes "I didn't want my gift to go to waste, so I joined the Knights.";

	next;

	mes "[Knight of Devotion]";
	mes "I respect others' opinions, but I refuse to bend mine to conform with others.";
	next;
	mes "[Knight of Devotion]";
	mes "Soon, a tourney will be held over an honorable position in the group.";
	next;
	mes "[Knight of Devotion]";

		next;
		mes "[Knight of Devotion]";
		mes "Are you sure that's your kind of justice?";
		next;
		if (select( "Yes.", "Actually, no." ) == 1)
			break;
	}

	if (.@faith)
		setquest 3433;
	else
		setquest 3432;
	job_ig_4th_answer = .@faith;
	mes "[Knight of Devotion]";
	mes "I see.";
	next;
	mes "[Knight of Devotion]";
	if (.@faith) {
		mes "I'm glad to meet a kindred spirit.";
		next;
		mes "[Knight of Devotion]";
		mes "Would you be in the tourney as well?";
		next;
		mes "[Knight of Devotion]";

		next;
		mes "[Knight of Devotion]";
		mes "I really want to carry our point.";
		next;
		mes "[Knight of Devotion]";
		mes "Thank you. Have safe travel back home.";
	} else {
		mes "Is focusing solely on force truly the right way...";
		next;
		mes "[Knight of Devotion]";
		mes "You hold different views than I do, but I respect that.";
		next;
		mes "[Knight of Devotion]";
		mes "Are you joining the tourney as well? Perhaps we'll meet in the combat ring.";
		next;
		mes "[Knight of Devotion]";
		mes "Well, I understand. Be safe on your travels.";
	}
	close;

	questinfo(QTYPE_JOB, QMARK_YELLOW, "isbegin_quest(3430) == 1");
	end;
